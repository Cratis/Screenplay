  compatible with scene.web
  compatible with Cratis.Components

layout ApplicationShell
  category application
  type masterDetail
  exposes selectedWorkItem WorkItemId
  outlet details
  navigation contributes Navigation
  content

ui profile Web
  target platform web
  target size expanded
  theme WorkspaceTheme
  layout ApplicationShell

  packages
    core
    scene.web
    Cratis.Components

  icons
    scene.icons
    workspace.icons

template ApplicationShell

exposure for WorkspaceFeatureShell
  property "shell:header".title label "Header title"
  property "shell:header".actions label "Header actions" operations add, remove, reorder, edit-fields fields label, icon

exposure for EditWorkItemDialog
  property "shell:header".title reexposes WorkspaceFeatureShell

instance WorkItemList
  set "shell:header".title = "Work items"
  items "shell:header".actions
    item "export:csv"
      label = "Export"
      icon = "download"
    item archive
      label = "Archive"
      icon = null

concept WorkItemId : Uuid
concept CommentId : Uuid
concept WorkStatus : Enum
  open
  closed

module Workspaces
  template WorkspaceFeatureShell

  screen template WorkspaceFeatureShell
    fits slot content
    category feature
    type masterDetail
    exposes selectedWorkItem WorkItemId
    outlet detail
    display "Workspace feature shell"
    description "A list with the selected item beside it"
    scopes module, feature, subfeature, slice
    header contributes WorkspaceActions
    list
    detail

    content header
      component scene.web.Header shellHeader
        id "shell:header"
        property title = "Workspace"
        property actions = []

    arrangement flow
      column gap 8
        header height 56
        grid gap 16 columns 2 rows 1 grow
          list grow 1 span 1
          detail grow 2 span 1

      when width compact
        column
          header
          row grow 1.5
            list grow

  dialog template EditWorkItemDialog
    category dialog
    type commandForm
    exposes submittedWorkItem WorkItemId
    display "Edit work item"
    scopes slice
    body
    actions

    content actions
      toolbar dialogDefaults
        item cancel navigate to WorkItemList

  form CreateWorkItemForm for CreateWorkItem
    generation manual
    field workItemId
    field title label "Title"
    field priority label "Priority"
    columns auto
    layout
      column 1 width 1fr min 240px max 50%
      column 2 width 2fr
      columnGap 24px
      rowGap 16px
      place workItemId row 1 column 1 width auto
      place title row 1 column 2 columnSpan 1 width 100%
      place priority row 2 column 1 columnSpan 2
    on submit navigate to WorkItemDetails by workItemId

  form RenameWorkItemForm for RenameWorkItem
    populate from item
    field workItemId
    field title label "New title"
    columns manual
      column title label "Title"
    on submit navigate to WorkItemDetails by workItemId

  feature Tracking
    template WorkspaceFeatureShell

    contribute to Navigation
      navigate to WorkItemList
      label "Work items"
      order 10
      id "nav:work-items"
      icon folder
      presentation primary
      group "Tracking"
      destination outlet details

    contribute to Navigation
      navigate to WorkItemList
      label "Edit work item"
      order 20
      id "nav:edit"
      group "Tracking"
      destination dialog EditWorkItemDialog

    contribute to Navigation
      navigate to WorkItemList
      label "Help"
      order 90
      destination external "https://cratis.io/help"

    slice StateChange CreateWorkItem
      command CreateWorkItem
        workItemId WorkItemId identifier
        title String
        priority Int
        produces WorkItemCreated
          for workItemId
          title = title
          priority = priority
          status = "open"

      event WorkItemCreated
        title String
        priority Int
        status WorkStatus

    slice StateChange RenameWorkItem
      command RenameWorkItem
        workItemId WorkItemId identifier
        title String
        produces WorkItemRenamed
          for workItemId
          title = title

      event WorkItemRenamed
        title String

    slice StateChange CloseWorkItem
      command CloseWorkItem
        workItemId WorkItemId identifier
        produces WorkItemClosed
          for workItemId
          status = "closed"

      event WorkItemClosed
        status WorkStatus

    slice StateView WorkItemList
      template WorkspaceFeatureShell

      readmodel WorkItemSummary
        workItemId WorkItemId
        title String
        priority Int
        status WorkStatus

      query AllWorkItems => WorkItemSummary[]
      query WorkItemById => WorkItemSummary optional
        by workItemId WorkItemId

      screen WorkItemList
        data WorkItemSummary[] via query AllWorkItems
        toolbar main
          item create action CreateWorkItem
            label "New work item"
            icon add
            presentation placement "primary"
          item refresh navigate to WorkItemList
            label "Clear selection"
            icon clear
            presentation route "work-items"
        component scene.web.DataGrid workItems
          id "work-items:list"
          context from query AllWorkItems
          property selectedItem from component "work-items:list".selectedItem null clear
          property initialSelection from literal null
          property title = "Work items"
          property pageSize = 25
          property selectable = true
          property emptyState = {"title":"No work items","actions":["create"]}
          property noSelection = null
          icon table
          presentation density "compact"
          exposes selectedWorkItem from component workItems.selectedItem.workItemId
          outlet detail
            summary selectedWorkItem
              field title label "Title"
              field status label "Status"
        contribute to WorkspaceActions order 10
          action CreateWorkItem
            navigate to WorkItemDetails
              outlet details
              parameter workItemId from component "work-items:list".selectedItem.workItemId
        contribute to Navigation
          section quickFilters
            title "Open items"

    slice StateView WorkItemDetails
      template WorkspaceFeatureShell

      readmodel WorkItemDetails
        workItemId WorkItemId
        title String
        priority Int
        status WorkStatus

      query GetWorkItem => WorkItemDetails optional
        by workItemId WorkItemId

      screen WorkItemDetails
        data WorkItemDetails via query GetWorkItem by workItemId
        toolbar detail
          item rename action RenameWorkItem
            label "Rename"
            icon edit
            presentation dialog "EditWorkItemDialog"
          item close action CloseWorkItem
            label "Close"
            icon close
            presentation confirm "Close this work item?"
          item deepLink navigate to WorkItemDetails
            parameter workItemId from data.workItemId
        component scene.web.DetailsPanel details
          id "details.panel:primary"
          context from data workItemId
          property title from data.title
          property status from data.status
          property pinned from literal false
          icon file
          presentation route "work-items/{workItemId}"
          outlet actions
            component scene.web.CommandForm renameForm
              context from data workItemId
              property initialTitle from data title null preserve
              exposes submittedWorkItem from data.workItemId

    slice StateView WorkItemDialog
      screen RenameWorkItemDialog
        template EditWorkItemDialog
          body
            component scene.web.CommandForm renameWorkItem
              context from data workItemId
              property workItemId from data workItemId
              property title from data title null preserve
          actions
            toolbar dialogActions
              item submit action RenameWorkItem
                label "Save"
                icon save
              item cancel navigate to WorkItemDetails
                parameter workItemId from data.workItemId
