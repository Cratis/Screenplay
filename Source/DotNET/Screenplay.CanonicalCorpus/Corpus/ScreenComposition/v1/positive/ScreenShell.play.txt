    item customers title "Customers" icon people target Sales.Customers

  feature Orders
    feature Returns

    slice StateChange PlaceOrder
      command PlaceOrder
        orderId Uuid identifier
        reason String
        produces OrderPlaced
          for orderId
          reason = reason

        reason String

    slice StateChange CancelOrder
      command CancelOrder
        orderId Uuid identifier
        reason String
        produces OrderCancelled
          for orderId
          reason = reason
      event OrderCancelled
        reason String

    slice StateView OrderList
