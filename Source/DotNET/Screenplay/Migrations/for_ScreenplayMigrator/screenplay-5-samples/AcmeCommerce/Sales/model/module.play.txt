// Sales: what a customer bought. Module skeleton (features are filled by the Sales slices).
module Orders
  description "Order intake, approval, fulfilment and reporting"
  authorize CanWorkOrders

  // The module adds these to the toolbar of every screen of its features: each runs one command of one of them, so they are always at hand.
  // Cancelling asks first.
  toolbar actions
    button approve title "Approve order" slice Orders.Decision.ApproveOrder emphasis primary icon check
    button cancel title "Cancel order" slice Orders.Fulfilment.CancelOrder emphasis quiet icon ban
      confirm "Cancel this order?"

  feature Intake
  feature Approval
  feature Decision
  feature Fulfilment
  feature Lookup
  feature Cancellations
  feature Reporting
