// Who may do what. Policies are the named rules; personas hold them by name.
// The event-model board draws a slice's screens on the row of every persona
// whose policies satisfy every gate in front of it - the module's, the
// feature's, and the slice's first command or query.

policy HasWorkspaceAccess
  require authenticated

policy IsConsultant
  require role "Consultant"

policy IsEngagementManager
  require role "EngagementManager"

policy IsPayrollOfficer
  require role "PayrollOfficer"

// Human payroll callers need the finance claim. The declared automation role is
// the trusted alternative: a system identity has no portable department claim.
policy IsFinanceDepartment
  require claim "department" matches "Finance" or role "PayrollAutomation"

persona Consultant
  description "Records the hours worked on each engagement and submits a timesheet every week"
  goal "Record hours as they are worked"
  goal "Submit the timesheet on Friday without chasing"
  policy HasWorkspaceAccess
  policy IsConsultant

persona EngagementManager
  description "Opens engagements, staffs them with consultants, and approves the hours logged against them"
  goal "Staff engagements with the right people"
  goal "Approve hours without opening every sheet"
  policy HasWorkspaceAccess
  policy IsEngagementManager

persona PayrollOfficer
  description "Turns approved hours into payroll runs and hands them to the external payroll provider"
  goal "Turn approved hours into a payroll run"
  goal "Hand the run to the payroll provider"
  policy HasWorkspaceAccess
  policy IsPayrollOfficer
  policy IsFinanceDepartment

authentication
  provider EntraId
